<выражение> ->
    <логическое выражение>

<логическое выражение> ->
    <логический одночлен> |
    <логический одночлен> <логическое выражение2> 

<логическое выражение2> ->
    ! <логический одночлен> <логическое выражение2>

<логический одночлен> ->
    <вторичное логическое выражение> |
    <вторичное логическое выражение> <логический одночлен2>

<логический одночлен2> ->
    <вторичное логическое выражение> |
    & <вторичное логическое выражение> <логический одночлен2>

<вторичное логическое выражение> ->
    <первичное логическое выражение> |
    ~ <первичное логическое выражение>

<первичное логическое выражение> ->
    <логическое значение> |
    <идентификатор>

<логическое значение> ->
    true | false

<знак логической операции> ->
    ~ | & | !

--------------------------------------------------

<программа> ->
    <блок>

<блок> ->
    { <список операторов> }

<список операторов>
    <оператор> <хвост>

<хвост> ->
    ; <оператор> <хвост> | ε

<оператор> ->
    <идентификатор> = <выражение>
